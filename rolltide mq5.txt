int Sx(const int v)
  {
   double s = InpScale;
   if(s < 0.6)
      s = 0.6;
   if(s > 2.0)
      s = 2.0;
   return (int)MathRound(v * s);
  }

int OnInit()
  {
   IndicatorSetString(INDICATOR_SHORTNAME, "ROLLTIDE");
   g_set.tide_tf    = InpTideTf;
   g_set.hour_tf    = InpHourTf;
   g_set.rip_tf     = InpRipTf;
   g_set.speed_tf   = InpSpeedTf;
   g_set.rsi_len    = InpRsiPeriod;
   g_set.sma_len    = InpSmaPeriod;
   g_set.shift      = InpShift;
   g_set.ma_method  = InpMaMethod;
   g_set.price      = InpPrice;
   g_set.cci_len    = InpCciPeriod;
   g_set.cci_price  = InpCciPrice;
   g_set.csv_name   = InpCsvName;
   RtPrepSet(g_set);
   g_px = InpFillWindow ? 0 : InpPanelX;
   g_py = InpFillWindow ? 0 : InpPanelY;
   g_focus = _Symbol;
   PaintPalette();
   ChartSetInteger(0, CHART_EVENT_MOUSE_MOVE, true);
   ApplyChartMode(true);
   if(!LoadWatch())
      Print("ROLLTIDE: no symbols in Market Watch — board will stay empty");
   ScanChartSymbol();
   EventSetMillisecondTimer(MathMax(80, InpTimerMs));
   Print("ROLLTIDE ready | ", g_n, " symbols | live ", RtPriceWord(InpPrice),
         " | rsi ", InpRsiPeriod, " sma ", InpSmaPeriod, " shift +", InpShift);
   Pulse(true);
   return INIT_SUCCEEDED;
  }

void OnDeinit(const int reason)
  {
   EventKillTimer();
   ChartSetInteger(0, CHART_EVENT_MOUSE_MOVE, false);
   ApplyChartMode(false);
   for(int i = 0; i < g_n; i++)
      RtReleaseRow(g_rows[i]);
   g_cv.Destroy();
   ObjectDelete(0, HUD_NAME);
   ChartRedraw(0);
  }

void ApplyChartMode(const bool attach)
  {
   if(attach)
     {
      g_prev_show = (ChartGetInteger(0, CHART_SHOW) != 0);
      g_hid_chart = InpHideChart;
      if(InpHideChart)
        {
         ChartSetInteger(0, CHART_SHOW, false);
         ChartSetInteger(0, CHART_SHOW_GRID, false);
         ChartSetInteger(0, CHART_SHOW_OHLC, false);
         ChartSetInteger(0, CHART_SHOW_BID_LINE, false);
         ChartSetInteger(0, CHART_SHOW_ASK_LINE, false);
         ChartSetInteger(0, CHART_SHOW_LAST_LINE, false);
         ChartSetInteger(0, CHART_SHOW_PERIOD_SEP, false);
         ChartSetInteger(0, CHART_SHOW_VOLUMES, CHART_VOLUME_HIDE);
         ChartSetInteger(0, CHART_SHOW_DATE_SCALE, false);
         ChartSetInteger(0, CHART_SHOW_PRICE_SCALE, false);
         ChartSetInteger(0, CHART_SHOW_ONE_CLICK, false);
         ChartSetInteger(0, CHART_COLOR_BACKGROUND, InpColBg);
         ChartSetInteger(0, CHART_COLOR_FOREGROUND, InpColBg);
        }
      return;
     }
   if(!g_hid_chart)
      return;
   ChartSetInteger(0, CHART_SHOW, g_prev_show);
   ChartSetInteger(0, CHART_SHOW_DATE_SCALE, true);
   ChartSetInteger(0, CHART_SHOW_PRICE_SCALE, true);
  }

void Pulse(const bool force)
  {
   const uint now = GetTickCount();
   if(!force && g_last_ms != 0 && (now - g_last_ms) < (uint)MathMax(200, InpTimerMs))
      return;
   g_last_ms = now;
   ScanAll();
   DrawHud();
  }

void OnChartEvent(const int id, const long &lparam, const double &dparam, const string &sparam)
  {
   if(id == CHARTEVENT_CHART_CHANGE)
     {
      const int w = (int)ChartGetInteger(0, CHART_WIDTH_IN_PIXELS);
      const int h = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
      if(MathAbs(w - g_cv_w) > 20 || MathAbs(h - g_cv_h) > 20)
         DrawHud();
      return;
     }
   if(id == CHARTEVENT_MOUSE_MOVE)
     {
      g_mx = (int)lparam;
      g_my = (int)dparam;
      HandleMouse(g_mx, g_my, sparam);
      return;
     }
   if(id == CHARTEVENT_CLICK)
     {
      HandleClick((int)lparam, (int)dparam);
      return;
     }
   if(id == CHARTEVENT_OBJECT_CLICK && sparam == HUD_NAME)
      HandleClick(g_mx, g_my);
  }

bool LoadWatch()
  {
   string raw[];
   int n = RtSplitList(InpSymbols, raw);
   if(n <= 0)
     {
      n = SymbolsTotal(true);
      ArrayResize(raw, n);
      int k = 0;
      for(int i = 0; i < n; i++)
        {
         string s = SymbolName(i, true);
         if(s != "")
           {
            raw[k] = s;
            k++;
           }
        }
      ArrayResize(raw, k);
      n = k;
     }

   const int cap = MathMax(1, MathMin(InpMaxSyms, RT_MAX_SYMS));
   ArrayResize(g_rows, cap);
   g_n = 0;
   g_chart = -1;
   for(int i = 0; i < n && g_n < cap; i++)
     {
      if(!RtBindRow(g_rows[g_n], raw[i], g_set))
        {
         Print("ROLLTIDE: cannot bind ", raw[i]);
         continue;
        }
      if(raw[i] == _Symbol)
         g_chart = g_n;
      g_n++;
     }
   ArrayResize(g_rows, g_n);
   return (g_n > 0);
  }

bool ScanChartSymbol()
  {
   if(g_chart >= 0)
      return false;
   if(g_n >= RT_MAX_SYMS)
      return false;
   ArrayResize(g_rows, g_n + 1);
   if(!RtBindRow(g_rows[g_n], _Symbol, g_set))
     {
      ArrayResize(g_rows, g_n);
      return false;
     }
   g_chart = g_n;
   g_n++;
   return true;
  }

void ScanAll()
  {
   for(int i = 0; i < g_n; i++)
     {
      if(g_focus != "" && g_rows[i].name == g_focus)
         g_chart = i;
      else if(g_focus == "" && g_rows[i].name == _Symbol)
         g_chart = i;
      const int before = g_rows[i].state;
      RtScoreRow(g_rows[i], g_set);
      if(!g_armed)
        {
         g_rows[i].prev_state = g_rows[i].state;
         continue;
        }
      if(RtIsFire(g_rows[i].state) && g_rows[i].state != before)
        {
         if(InpWriteCsv)
            RtWriteSignal(g_rows[i], InpCsvName);
         if(InpAlerts)
            FireAlert(g_rows[i]);
        }
      g_rows[i].prev_state = g_rows[i].state;
     }
   g_armed = true;
  }

void FireAlert(const SRollRow &row)
  {
   const string side = RtSideWord(row.state);
   const string msg = "ROLLTIDE  " + row.name + "  " + side
                      + "  " + RtTfLabel(g_set.speed_tf) + " CCI "
                      + DoubleToString(row.cci_lock, 1);
   Alert(msg);
   if(InpAlertSound != "")
      PlaySound(InpAlertSound);
   Print(msg);
  }

bool WantRow(const SRollRow &row)
  {
   return RtIsFire(row.state);
  }

void DrawHud()
  {
   if(!InpShowGlass && !InpShowBoard)
      return;

   SRollRow view[];
   ArrayResize(view, MathMax(1, g_n));
   int shown = 0;
   for(int i = 0; i < g_n; i++)
     {
      if(!WantRow(g_rows[i]))
         continue;
      view[shown] = g_rows[i];
      shown++;
     }
   ArrayResize(view, shown);
   RtSortRows(view, shown);

   int chart_w = (int)ChartGetInteger(0, CHART_WIDTH_IN_PIXELS);
   int chart_h = (int)ChartGetInteger(0, CHART_HEIGHT_IN_PIXELS);
   if(chart_w < 400)
      chart_w = (g_cv_w >= 400 ? g_cv_w : 1200);
   if(chart_h < 280)
      chart_h = (g_cv_h >= 280 ? g_cv_h : 720);
   const int pad = Sx(22);
   const int brandH = Sx(26);
   const int headH = Sx(28);
   const int rowH = Sx(36);
   const int innerW = InpFillWindow ? MathMax(Sx(520), chart_w - pad * 2) : Sx(520);
   g_head_h = pad + brandH + Sx(10);

   int max_rows = InpMaxRows;
   int h = InpFillWindow ? chart_h : (g_head_h + Sx(4));
   int w = InpFillWindow ? chart_w : (pad * 2 + innerW);
   if(!InpFillWindow)
     {
      const int fit = MathMax(1, (chart_h - g_py - h - pad) / rowH);
      max_rows = MathMax(1, MathMin(InpMaxRows, fit));
      h += headH + MathMax(1, MathMin(shown, max_rows)) * rowH + pad;
     }
   else
     {
      const int fit = MathMax(1, (chart_h - g_head_h - headH - pad) / rowH);
      max_rows = MathMax(1, MathMin(InpMaxRows, fit));
     }

   if(!EnsureCanvas(w, h))
      return;

   g_cv.Erase(C_BG);
   g_cv.FillRectangle(0, 0, Sx(4), h, C_ACCENT);

   Font(InpFont, 16, FW_BOLD);
   Text(pad, pad, Sx(220), brandH, "ROLLTIDE", C_ACCENT, TA_LEFT);
   Font(InpFont, 12, FW_BOLD);
   Text(w - pad - Sx(180), pad, Sx(180), brandH,
        IntegerToString(shown) + " SIGNALS", shown > 0 ? C_LONG : C_MUTE, TA_RIGHT);

   const int gx = pad;
   const int cy = g_head_h;
   const int c0 = gx;
   const int c1 = gx + innerW * 10 / 100;
   const int c2 = gx + innerW * 40 / 100;
   const int c3 = gx + innerW * 70 / 100;
   const int w0 = c1 - c0;
   const int w1 = c2 - c1;
   const int w2 = c3 - c2;
   const int w3 = gx + innerW - c3;

   FillRound(gx, cy, innerW, headH, Sx(4), C_HEAD);
   Font(InpFont, 10, FW_BOLD);
   Text(c0, cy, w0, headH, "#", C_MUTE, TA_CENTER);
   Text(c1 + Sx(8), cy, w1, headH, "SYMBOL", C_MUTE, TA_LEFT);
   Text(c2, cy, w2, headH, "SIDE", C_MUTE, TA_CENTER);
   Text(c3, cy, w3, headH, "CCI", C_MUTE, TA_CENTER);

   g_hits = 0;
   ArrayResize(g_hit_sym, InpMaxRows + 2);
   ArrayResize(g_hit_y0, InpMaxRows + 2);
   ArrayResize(g_hit_y1, InpMaxRows + 2);

   const int rows = MathMax(1, MathMin(shown, max_rows));
   if(shown == 0)
     {
      Font(InpFont, 12, FW_BOLD);
      Text(gx, cy + headH, innerW, rowH, "NO SIGNALS", C_MUTE, TA_CENTER);
     }
   else
     {
      for(int r = 0; r < rows; r++)
        {
         const int ry = cy + headH + r * rowH;
         const bool lng = (view[r].state == RT_FIRE_LONG);
         if((r % 2) == 1)
            g_cv.FillRectangle(gx, ry + 1, gx + innerW, ry + rowH - 2, C_HEAD);
         FillRound(c2 + (w2 - Sx(88)) / 2, ry + Sx(6), Sx(88), rowH - Sx(12), Sx(6),
                   lng ? C_LONG_FILL : C_SHORT_FILL);
         string num = IntegerToString(r + 1);
         if(r + 1 < 10)
            num = "0" + num;
         Font(InpFont, 12, FW_BOLD);
         Text(c0, ry, w0, rowH, num, C_ACCENT, TA_CENTER);
         Text(c1 + Sx(8), ry, w1, rowH, view[r].name, C_TEXT, TA_LEFT);
         Text(c2, ry, w2, rowH, lng ? "LONG" : "SHORT", lng ? C_LONG : C_SHORT, TA_CENTER);
         Text(c3, ry, w3, rowH, RtSigned(view[r].cci_live, 1), lng ? C_LONG : C_SHORT, TA_CENTER);
         if(g_hits < ArraySize(g_hit_sym))
           {
            g_hit_sym[g_hits] = view[r].name;
            g_hit_y0[g_hits] = ry;
            g_hit_y1[g_hits] = ry + rowH;
            g_hits++;
           }
        }
     }

   g_cv.Update(false);
  }

void PaintCard(const int x, const int y, const int w, const int h,
               const string eye, const string stack, const string body,
               const bool ready, const int side, const double rsi, const double delta)
  {
   FillRound(x + 3, y + 4, w, h, Sx(10), C_SHADOW);
   StrokeRound(x, y, w, h, Sx(10), C_LINE, C_CARD);
   Font(InpFont, 11, FW_BOLD);
   Text(x + Sx(14), y + Sx(10), w - Sx(28), Sx(18), eye, C_ACCENT, TA_LEFT);
   Font(InpFont, 10, FW_NORMAL);
   Text(x + Sx(14), y + Sx(30), w - Sx(28), Sx(16), stack, C_MUTE, TA_LEFT);

   const int bx = x + Sx(10);
   const int by = y + Sx(52);
   const int bw = w - Sx(20);
   const int bh = InpShowValues ? Sx(48) : Sx(32);
   uint fill = C_IDLE;
   uint ink = C_MUTE;
   if(ready && side == RT_ABOVE)
     {
      fill = C_LONG_FILL;
      ink = C_LONG;
     }
   else if(ready && side == RT_BELOW)
     {
      fill = C_SHORT_FILL;
      ink = C_SHORT;
     }
   g_cv.FillRectangle(bx, by, bx + bw, by + bh, fill);
   Font(InpFont, ready && side != RT_DEAD ? 14 : 11, FW_BOLD);
   if(InpShowValues && ready)
     {
      Text(bx, by + Sx(2), bw, Sx(22), body, ink, TA_CENTER);
      Font(InpFont, 9, FW_NORMAL);
      Text(bx, by + Sx(22), bw, Sx(22),
           DoubleToString(rsi, 1) + "   " + RtSigned(delta, 1), C_MUTE, TA_CENTER);
     }
   else
      Text(bx, by, bw, bh, body, ink, TA_CENTER);
  }

void PaintSpeed(const int x, const int y, const int w, const int h, const SRollRow &row)
  {
   FillRound(x + 3, y + 4, w, h, Sx(10), C_SHADOW);
   StrokeRound(x, y, w, h, Sx(10), C_LINE, C_CARD);
   Font(InpFont, 11, FW_BOLD);
   Text(x + Sx(14), y + Sx(10), w - Sx(28), Sx(18), "SPEED", C_ACCENT, TA_LEFT);
   Font(InpFont, 10, FW_NORMAL);
   Text(x + Sx(14), y + Sx(30), w - Sx(28), Sx(16),
        RtTfLabel(g_set.speed_tf) + "  CCI " + IntegerToString(g_set.cci_len), C_MUTE, TA_LEFT);

   const int bx = x + Sx(10);
   const int by = y + Sx(52);
   const int bw = w - Sx(20);
   const int bh = InpShowValues ? Sx(48) : Sx(32);
   string body = WaitText(RtTfLabel(g_set.speed_tf), row.have_cci, g_set.need_cci);
   uint ink = C_MUTE;
   if(row.ready)
     {
      body = RtSigned(row.cci_live, 1);
      if(row.cci_live > 0.0)
         ink = C_LONG;
      else if(row.cci_live < 0.0)
         ink = C_SHORT;
     }
   g_cv.FillRectangle(bx, by, bx + bw, by + bh, C_IDLE);
   Font(InpFont, 14, FW_BOLD);
   Text(bx, by, bw, InpShowMeter && row.ready ? Sx(26) : bh, body, ink, TA_CENTER);
   if(InpShowMeter && row.ready)
      PaintZeroBar(bx + Sx(8), by + bh - Sx(14), bw - Sx(16), Sx(8), row.cci_live);
  }

void PaintBanner(const int x, const int y, const int w, const int h, const SRollRow &row)
  {
   uint fill = C_HEAD;
   uint ink = C_MUTE;
   if(row.ready && row.state == RT_FIRE_LONG)
     {
      fill = C_LONG_FILL;
      ink = C_LONG;
     }
   else if(row.ready && row.state == RT_FIRE_SHORT)
     {
      fill = C_SHORT_FILL;
      ink = C_SHORT;
     }
   else if(row.ready && (row.state == RT_WAIT_LONG || row.state == RT_WAIT_SHORT))
     {
      fill = C_GOLD_FILL;
      ink = C_GOLD;
     }
   FillRound(x, y, w, h, Sx(8), fill);
   Font(InpFont, 12, FW_BOLD);
   Text(x, y, w, h, RtBanner(row, g_set), ink, TA_CENTER);
  }

void PaintBoardRow(const int x, const int y, const int w, const int h, const SRollRow &row, const int rank,
                   const int c0, const int c1, const int c2, const int c3, const int c4, const int c5,
                   const int w0, const int w1, const int w2, const int w3, const int w4, const int w5)
  {
   string num = IntegerToString(rank);

void HandleMouse(const int cx, const int cy, const string flags)
  {
   if(!InpDragPanel || InpFillWindow)
      return;
   const bool down = (StringFind(flags, "1") >= 0);
   int lx, ly;
   LocalXY(cx, cy, lx, ly);
   if(!g_drag && down && lx >= 0 && ly >= 0 && lx < g_cv_w && ly < g_head_h)
     {
      g_drag = true;
      g_drag_x = cx;
      g_drag_y = cy;
      return;
     }
   if(g_drag && down)
     {
      const int dx = cx - g_drag_x;
      const int dy = cy - g_drag_y;
      g_drag_x = cx;
      g_drag_y = cy;
      if(InpCorner == CORNER_RIGHT_UPPER || InpCorner == CORNER_RIGHT_LOWER)
         g_px -= dx;
      else
         g_px += dx;
      if(InpCorner == CORNER_LEFT_LOWER || InpCorner == CORNER_RIGHT_LOWER)
         g_py -= dy;
      else
         g_py += dy;
      if(g_px < 0) g_px = 0;
      if(g_py < 0) g_py = 0;
      EnsureCanvas(g_cv_w, g_cv_h);
      return;
     }
   g_drag = false;
  }

void PaintZeroBar(const int x, const int y, const int w, const int h, const double cci)
  {
   if(w < 8 || h < 3)
      return;
   g_cv.FillRectangle(x, y, x + w, y + h, C_IDLE);
   const int mid = x + w / 2;
   g_cv.Line(mid, y, mid, y + h, C_ACCENT);
   double scale = InpCciMeter;
   if(scale < 20.0)
      scale = 20.0;
   double pct = MathAbs(cci) / scale;
   if(pct > 1.0)
      pct = 1.0;
   const int span = MathMax(1, (int)MathRound((w / 2 - 1) * pct));
   uint ink = C_MUTE;
   if(cci > 0.0)
      ink = C_LONG;
   else if(cci < 0.0)
      ink = C_SHORT;
   if(cci >= 0.0)
      g_cv.FillRectangle(mid, y + 1, mid + span, y + h - 1, ink);
   else
      g_cv.FillRectangle(mid - span, y + 1, mid, y + h - 1, ink);
  }

void FillRound(const int x, const int y, const int w, const int h, const int r, const uint clr)
  {
   if(w < 2 || h < 2)
      return;
   int rad = r;
   if(rad * 2 > w)
      rad = w / 2;
   if(rad * 2 > h)
      rad = h / 2;
   if(rad < 1)
     {
      g_cv.FillRectangle(x, y, x + w - 1, y + h - 1, clr);
      return;
     }
   g_cv.FillRectangle(x + rad, y, x + w - 1 - rad, y + h - 1, clr);
   g_cv.FillRectangle(x, y + rad, x + rad, y + h - 1 - rad, clr);
   g_cv.FillRectangle(x + w - 1 - rad, y + rad, x + w - 1, y + h - 1 - rad, clr);
   g_cv.FillCircle(x + rad, y + rad, rad, clr);
   g_cv.FillCircle(x + w - 1 - rad, y + rad, rad, clr);
   g_cv.FillCircle(x + rad, y + h - 1 - rad, rad, clr);
   g_cv.FillCircle(x + w - 1 - rad, y + h - 1 - rad, rad, clr);
  }

void StrokeRound(const int x, const int y, const int w, const int h,

int RtSplitList(const string text, string &out[])
  {
   string t = text;
   StringTrimLeft(t);
   StringTrimRight(t);
   if(t == "")
     {
      ArrayResize(out, 0);
      return 0;
     }
   string parts[];
   int n = StringSplit(t, ',', parts);
   int k = 0;
   ArrayResize(out, n);
   for(int i = 0; i < n; i++)
     {
      StringTrimLeft(parts[i]);
      StringTrimRight(parts[i]);
      if(parts[i] != "")
        {
         out[k] = parts[i];
         k++;
        }
     }
   ArrayResize(out, k);
   return k;
  }

bool RtReadPair(const int h_rsi, const int h_ma, const int need,
                double &now, double &shadow, int &have)
  {
   now = 0.0;
   shadow = 0.0;
   const int a = BarsCalculated(h_rsi);
   const int b = BarsCalculated(h_ma);
   have = MathMin(a, b);
   if(have < need)
      return false;
   double rsi[], ma[];
   ArraySetAsSeries(rsi, true);
   ArraySetAsSeries(ma, true);
   if(CopyBuffer(h_rsi, 0, 0, 1, rsi) < 1)
      return false;
   if(CopyBuffer(h_ma, 0, 0, 1, ma) < 1)
      return false;
   if(rsi[0] == EMPTY_VALUE || ma[0] == EMPTY_VALUE)
      return false;
   if(!MathIsValidNumber(rsi[0]) || !MathIsValidNumber(ma[0]))
      return false;
   now = rsi[0];
   shadow = ma[0];
   return true;
  }

bool RtReadCci(const int handle, const int need, double &cci, int &have)
  {
   cci = 0.0;
   have = BarsCalculated(handle);
   if(have < need)
      return false;
   double buf[];
   ArraySetAsSeries(buf, true);
   if(CopyBuffer(handle, 0, 0, 1, buf) < 1)
      return false;
   if(buf[0] == EMPTY_VALUE || !MathIsValidNumber(buf[0]))
      return false;
   cci = buf[0];
   return true;
  }

void RtScoreRow(SRollRow &row, const SRollSet &set)
  {
   double d1_now, d1_sh, h1_now, h1_sh, m5_now, m5_sh, cci;
   const bool ok_d1 = RtReadPair(row.h_d1, row.h_d1m, set.need_rsi, d1_now, d1_sh, row.have_d1);
   const bool ok_h1 = RtReadPair(row.h_h1, row.h_h1m, set.need_rsi, h1_now, h1_sh, row.have_h1);
   const bool ok_m5 = RtReadPair(row.h_m5, row.h_m5m, set.need_rsi, m5_now, m5_sh, row.have_m5);
   const bool ok_cci = RtReadCci(row.h_cci, set.need_cci, cci, row.have_cci);

   row.ready = (ok_d1 && ok_h1 && ok_m5 && ok_cci);
   if(!row.ready)
     {
      row.d1 = RT_DEAD;
      row.h1 = RT_DEAD;
      row.m5 = RT_DEAD;
      row.state = RT_FLAT;
      return;
     }

   row.rsi_d1 = d1_now;
   row.rsi_h1 = h1_now;
   row.rsi_m5 = m5_now;
   row.sh_d1 = d1_sh;
   row.sh_h1 = h1_sh;
   row.sh_m5 = m5_sh;
   row.cci_live = cci;
   row.d1 = RtClock(d1_now, d1_sh);
   row.h1 = RtClock(h1_now, h1_sh);
   row.m5 = RtClock(m5_now, m5_sh);
   row.state = RtState(row.d1, row.h1, row.m5);

   if(RtIsFire(row.state) && !row.locked)
     {
      row.cci_lock = cci;
      row.locked = true;
     }
   if(!RtIsFire(row.state))
      row.locked = false;
  }

bool RtWorse(const SRollRow &a, const SRollRow &b)
  {
   const double da = MathAbs(RtSortCci(a));
   const double db = MathAbs(RtSortCci(b));
   if(da != db)
      return (da < db);
   return (RtBucket(a.state) > RtBucket(b.state));
  }

void RtSortRows(SRollRow &rows[], const int n)
  {
   for(int i = 0; i < n; i++)
     {
      for(int j = i + 1; j < n; j++)
        {
         if(RtWorse(rows[i], rows[j]))
           {
            SRollRow tmp = rows[i];
            rows[i] = rows[j];
            rows[j] = tmp;
           }
        }
     }
  }

void RtWriteSignal(const SRollRow &row, const string file_name)
  {
   const string side = RtSideWord(row.state);
   if(side == "" || file_name == "")
      return;
   const int flags = FILE_READ | FILE_WRITE | FILE_TXT | FILE_ANSI | FILE_SHARE_READ | FILE_SHARE_WRITE;
   int h = FileOpen(file_name, flags);
   if(h == INVALID_HANDLE)
      h = FileOpen(file_name, FILE_WRITE | FILE_TXT | FILE_ANSI);
   if(h == INVALID_HANDLE)
      return;
   if(FileSize(h) == 0)
      FileWriteString(h, "time,symbol,side,cci,rsi_tide,rsi_hour,rsi_rip\n");
   FileSeek(h, 0, SEEK_END);
   string line = TimeToString(TimeLocal(), TIME_DATE | TIME_SECONDS)
                 + "," + row.name
                 + "," + side
                 + "," + DoubleToString(row.cci_lock, 2)
                 + "," + DoubleToString(row.rsi_d1, 2)
                 + "," + DoubleToString(row.rsi_h1, 2)
                 + "," + DoubleToString(row.rsi_m5, 2)
                 + "\n";
   FileWriteString(h, line);
   FileClose(h);
  }
